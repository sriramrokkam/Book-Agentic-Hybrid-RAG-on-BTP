% ── H1 / H2 / H3 heading hierarchy ──────────────────────────────────
% H1 — Chapter title: 22pt, SAP Blue, bold, extra space above
% H2 — Section:       16pt, SAP Blue, bold
% H3 — Subsection:    13pt, SAP Blue 80%, bold
% H4 — Subsubsection: 11pt, SAP Blue 60%, bold italic
\usepackage{titlesec}

\titleformat{\section}
  {\fontspec{72 Brand Bold}\fontsize{22pt}{26pt}\selectfont\color{sap-blue}}
  {}{0pt}{}
\titlespacing{\section}{0pt}{32pt}{12pt}

\titleformat{\subsection}
  {\fontspec{72 Brand Bold}\fontsize{16pt}{20pt}\selectfont\color{sap-blue}}
  {\thesubsection}{0.8em}{}
\titlespacing{\subsection}{0pt}{20pt}{8pt}

\titleformat{\subsubsection}
  {\fontspec{72 Brand Bold}\fontsize{13pt}{17pt}\selectfont\color{sap-blue!80}}
  {\thesubsubsection}{0.8em}{}
\titlespacing{\subsubsection}{0pt}{14pt}{6pt}

\titleformat{\paragraph}[runin]
  {\fontspec{72 Brand Bold}\fontsize{11pt}{14pt}\selectfont\color{sap-blue!60}\itshape}
  {}{0pt}{}[.\quad]
